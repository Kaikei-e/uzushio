// Function to find the nth number 
int nextNumber(int n) 
{ 
    // Base case 
    if (n == 0) 
        return 0; 
    // Recursive case 
    else if (n == 1) 
        return -1; 
    else if (n == 2) 
        return 1; 
    else 
        return nextNumber(n - 1) + nextNumber(n - 2); 
} 

// Driver program 
int main() 
{ 
    int n; 
    cout << "Enter the value of n: "; 
    cin >> n; 
    cout << "The nth number is: " << nextNumber(n); 
    return 0; 
}
